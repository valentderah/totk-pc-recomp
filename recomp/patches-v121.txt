site fps60 main 8f27f4 6b08015f 7100055f   # raise path: cmp w10, #1

site fps60 main 8f2858 b9400288 52800028   # lower path: mov w8, #1 (minimum interval 1)
site fps60 main 8f2874 2a0a03f5 1a9fa555   # lower path: csinc w21, w10, wzr, ge (result never below 1)
