# each original word checked against the 1.4.3 main. MacNativeGames branches to veneers after the text; here a
# veneer's body is the site's replacement. The patch page holds three floats the runtime writes at every present
# (frame_rate.c): +0 the 30 Hz ratio (seconds * 30), +4 the frame step in seconds, +8 the step in 60 Hz ticks.

# 60 fps: the adaptive present-interval controller (main+0x2db174, 1.2.1 0x8f26c8) gets the minimum interval 1.
site fps60 main 2db204 6b08015f 7100055f   # clamp: cmp w10, #1 (was cmp w10, w8 = minimum)
site fps60 main 2db208 1a89b10a 1a9fa52a   # clamp: csinc w10, w9, wzr, ge (desired = max(desired, 1))
site fps60 main 2db22c b940468a d503201f   # lower trigger: nop the stored-candidate load
site fps60 main 2db360 6b0801bf 710005bf   # raise path: cmp w13, #1
site fps60 main 2db36c 1a88c1ac 1a9fc5ac   # raise path: csinc w12, w13, wzr, gt (next lower candidate floors at 1)
site fps60 main 2db340 b9400288 52800028   # lower path: mov w8, #1 (minimum interval 1)
site fps60 main 2db34c 2a0a03f5 1a9fa555   # lower path: csinc w21, w10, wzr, ge (result never below 1)

# Real-time step. The frame-step producer feeds both FrameRateMgr call sites with s8 = nominal interval * a 0.5/1.0
# coefficient (sys+0x1da bit 0, fcsel above the site); the nominal interval comes from the patch page, the coefficient
# and FrameRateMgr's own arithmetic stay. x8 is free: the ldxr after the site loads it. The global ratio after the
# controller call (1.2.1 0x8f2394) reads the ratio from the page instead of 60 / interval * 0.5.
site realtime main 3e5658 1e210808 @page:x8 bd400900 1e210808   # fmul s8, s0, s1: s0 = [page+8] first
site realtime main 2daa54 bd425260 @page:x9   # ldr s0, [x19, #0x250] -> adrp x9, page
site realtime main 2daa58 52a84e08 bd400120   # mov w8, #60.0 -> ldr s0, [x9] (global ratio)
site realtime main 2daa6c 5e21d800 d503201f   # scvtf s0, s0 -> nop
site realtime main 2daa70 1e201820 d503201f   # fdiv s0, s1, s0 -> nop
site realtime main 2daa74 1e2c1001 d503201f   # fmov s1, #0.5 -> nop
site realtime main 2daa78 1e210800 d503201f   # fmul s0, s0, s1 -> nop

# Hair and cloth: the actor animation update builds a second clock {seconds = factor / 30, ratio = factor} per
# rendered frame. 1.4.3 inlines it in two actor paths, where the factor was just stored to the ratio slot: the
# replacement stores measured ratio * factor there and computes measured seconds * factor. Two other callers share the
# out-of-line update (main+0x2a0e40), whose s2 (seconds) and s3 (ratio) are scaled by the measured ratio; its third
# caller already passes measured time and stays. x8 and s4 are free at all four sites.
site cloth main 246eac 1e200940 @page:x8 bd400100 1e200940 bd0367e0 bd400500 1e200940   # fmul s0, s10, s0
site cloth main 345904 1e210801 @page:x8 bd400101 1e210801 bd0117e1 bd400501 1e210801   # fmul s1, s0, s1
site cloth main 24944c 94015e7d @page:x8 bd400104 1e240842 1e240863 94015e7d   # bl 0x2a0e40
site cloth main 28bc98 9400546a @page:x8 bd400104 1e240842 1e240863 9400546a   # bl 0x2a0e40

# Trampolines: the contact code scales the bounce vector by 0.1 when time object +0x28 (30 / (ratio * time scale)) is
# over 100; compare +0x28 * +0x38 (30 / time scale) instead, the same value at 30 Hz and independent of the frame rate.
# s1 is set again by the fmov two instructions on.
site bounce main 562a64 bd402920 bd402920 bd403921 1e210800   # ldr s0, [x9, #0x28]; ldr s1, [x9, #0x38]; fmul s0, s0, s1

# Fades (MacNativeGames game_patches_v143.c patch_fade_increment, upstream 3e9e7f67): the distance/LOD mesh fades and
# the pre-actor fades add trunc(2.1 * ratio) to a counter each frame: 2 at 30 fps, 1 at 60, 0 above about 63.5 fps
# (ratio below 1 / 2.1), where a fade stops and its object stays translucent or invisible until a slower frame moves
# it on. Each site floors the increment at 1 without the flags: clz is 32 only for 0, so increment += clz >> 5. The
# scratch register is one the next instruction writes (w10 at 0x0a7ac4: ldrb w10; w11 at the pre-actor sites: add
# w11), or at 0x95dd24 a temporary the function no longer reads (add, strb, return).
set fade     TOTK_HIGHFPS_FIX    1
site fade main 0a7ac4 1e380008 1e380008 5ac0110a 53057d4a 0b0a0108   # fcvtzs w8, s0; clz w10, w8; lsr w10, w10, #5; add w8, w8, w10
site fade main 95dd24 1e380008 1e380008 5ac0110a 53057d4a 0b0a0108   # the same, second mesh fade
site fade main 6155e4 1e38000a 1e38000a 5ac0114b 53057d6b 0b0b014a   # fcvtzs w10, s0; clz w11, w10; lsr w11, w11, #5; add w10, w10, w11
site fade main 615660 1e38000a 1e38000a 5ac0114b 53057d6b 0b0b014a   # the same, second pre-actor fade path

# Camera aspect ratio for a game resolution that is not 16:9 (upstream game_patches_v143.c and aspect_patch.h
# aspect_install / aspect_stub; runtime/resolution.c res_init writes the output's width / height to page +56 and
# switches the set on). Each site leaves the ratio sead's perspective projection (+0xb0) takes in s0: two camera sites
# (update and setup) load the camera's copy of DefaultAspect 1.777778, two viewport routines divide width by height.
# The replacement loads the ratio from the page instead, so a 21:9 output shows more of the world at the sides with
# the vertical view kept (Hor+). x10 is kept on the stack, so no register liveness is assumed.

# The game's zstd decoder (BSD zstd, statically linked into main) through the host's libzstd (runtime/ov_zstd.c):
# every call of ZSTD_decompressMultiFrame (main+0x3a1f0: dctx, dst, capacity, src, size, dict, dict size, ddict) and of
# its wrapper ZSTD_decompress_usingDDict (main+0x3a1e0, which tail-branches to it) goes to a host function on the patch
# page (+0xff0 and +0xfe0). The lifted decoder runs its copy loops as chains of small functions; the host decodes the
# same frames several times faster and gives the virtual core up for large inputs. Not main+0x547a04: that caller
# (main+0x547964) resets a static DCtx of its own with which the game's zstd rejects some valid frames
# (corruption_detected on a 473 KB frame the host decodes), and the game relies on that result.
set zstd     TOTK_ZSTD_NATIVE    1
